    ENROLL_CHECK -.->|3 Days Expired| SEAT_EXPIRED

    SEAT_EXPIRED ==>|Release Vacancy| WAIT_CALL
    WAITING_LIST_10 ==> WAIT_CALL
    WAIT_CALL ==>|Admitted| FINAL_ADMITTED

    %% Step 4: Strict Branch + Subject Batch Allocation

    BRANCH_SUB_LOCK ==>|Assign Specific Batch| MEM_LINK
    MEM_LINK ==>|Connect to Active Cohort| BATCH_RUN

    WS_DISTRIBUTE -.-> FEE2
    FINAL_ADMITTED -.-> FEE3

    FEE2 -.-> PAY_GW_ROUTER
    FEE3 -.-> PAY_GW_ROUTER
    FEE4 -.-> PAY_GW_ROUTER

    PAY_GW_ROUTER --> WEBHOOK_VAL
    WEBHOOK_VAL --> AUTO_REC
    WEBHOOK_VAL --> LEDGER_POST
